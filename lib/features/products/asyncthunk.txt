import { createSlice, createAsyncThunk, PayloadAction } from '@reduxjs/toolkit';

export interface CheckoutItem {
  idList: string;
  name: string;
  price: number;
  quantity: number;
}

export interface ProductState {
  items: CheckoutItem[];
  loading: boolean;
  error: string | null;
}

// 1. Definisikan initialState awal yang kosong
const initialState: ProductState = {
  items: [],
  loading: false,
  error: null,
};

// 2. Buat Async Thunk untuk membaca file JSON secara asinkron (simulasi fetch DB)
export const fetchProductsFromJSON = createAsyncThunk(
  'products/fetchProducts',
  async (_, { rejectWithValue }) => {
    try {
      // Mengambil file JSON dari folder public aplikasi
      const response = await fetch('/products.json'); 
      if (!response.ok) {
        throw new Error('Gagal mengambil data dari database JSON');
      }
      const data = await response.json();
      return data as CheckoutItem[];
    } catch (error: any) {
      return rejectWithValue(error.message);
    }
  }
);

export const productSlice = createSlice({
  name: 'products',
  initialState,
  reducers: {
    removeItem: (state, action: PayloadAction<string>) => {
      state.items = state.items.filter(item => item.idList !== action.payload);
    },
  },
  // 3. Tangani hasil thunk asinkron di extraReducers
  extraReducers: (builder) => {
    builder
      .addCase(fetchProductsFromJSON.pending, (state) => {
        state.loading = true;
        state.error = null;
      })
      .addCase(fetchProductsFromJSON.fulfilled, (state, action) => {
        state.loading = false;
        state.items = action.payload; // Mengisi state dengan data dari file JSON
      })
      .addCase(fetchProductsFromJSON.rejected, (state, action) => {
        state.loading = false;
        state.error = action.payload as string;
      });
  },
});

export const { removeItem } = productSlice.actions;
export default productSlice.reducer;
